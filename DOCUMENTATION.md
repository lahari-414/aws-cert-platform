# AWS Cloud Practitioner Certification Platform
## Complete Project Documentation

---

## 1. PROJECT OVERVIEW

This is a full web-based AWS Certification Learning Platform built using only HTML, CSS, and JavaScript — no frameworks, no paid tools. It allows a user to:

1. Watch 5 video lessons (with real audio narration using browser Text-to-Speech)
2. Fill a registration form (name, qualification, email)
3. Take a 10-question AI-generated exam
4. Score 8 or more to pass → download an official-style certificate with their name
5. Score below 8 → get a "not certified" popup

```javascript
function playVideo() {
  videoPlaying = true;
  document.getElementById("playPauseBtn").textContent = "⏸ Pause";

  // Start progress bar timer (counts 180 seconds = 3 minutes)
  videoTimerInterval = setInterval(() => {
    videoSeconds++;
    const pct = Math.min((videoSeconds / VIDEO_DURATION) * 100, 100);
    document.getElementById("fakeProgFill").style.width = pct + "%";

    if (videoSeconds >= 30) { // Demo: 30 seconds instead of 180
      clearInterval(videoTimerInterval);
      videoCompleted();
    }
  }, 1000);

  // Start speaking
  speakNextLine();
}

function speakNextLine() {
  if (!videoPlaying) return;
  
  const line = videoScripts[currentVideoId][scriptIndex];
  scriptIndex++;

  // Web Speech API — built into every browser, completely free
  const utterance = new SpeechSynthesisUtterance(line);
  utterance.rate   = 0.92;  // slightly slower than normal
  utterance.pitch  = 1.0;
  utterance.volume = 1.0;

  // Choose best available English voice
  const voices   = speechSynthesis.getVoices();
  const preferred = voices.find(v =>
    (v.name.includes("Google") || v.name.includes("Microsoft")) &&
    v.lang.startsWith("en")
  );
  if (preferred) utterance.voice = preferred;

  // When this sentence finishes, speak the next one
  utterance.onend = () => {
    if (videoPlaying) speakNextLine();
  };

  speechSynthesis.speak(utterance);
}
```

#### Section E: AI Agent — Google Gemini API Call

```javascript
const GEMINI_API_KEY = "YOUR_GEMINI_API_KEY_HERE"; // Free key from aistudio.google.com

async function generateQuestionsFromAI() {
  // Show loading spinner
  document.getElementById("aiLoader").style.display = "block";

  try {
    // Call Gemini AI API
    const response = await fetch(
      `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.0-flash:generateContent?key=${GEMINI_API_KEY}`,
      {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({
          contents: [{
            parts: [{
              text: `Generate exactly 10 AWS Cloud Practitioner exam questions.
                     Cover: Cloud Concepts, EC2, S3, Lambda, IAM, Pricing, Regions.
                     Return ONLY raw JSON array:
                     [{"question":"...","options":["A","B","C","D"],"correctIndex":1}]`
            }]
          }],
          generationConfig: { temperature: 0.8, maxOutputTokens: 3000 }
        })
      }
    );

    const data    = await response.json();
    let   aiText  = data.candidates[0].content.parts[0].text;

    // Clean up any markdown formatting AI might add
    aiText = aiText.replace(/```json/g, "").replace(/```/g, "").trim();

    // Parse the JSON array of questions
    questions = JSON.parse(aiText);

  } catch(err) {
    // If API fails for any reason, use 10 hardcoded backup questions
    console.warn("AI failed, using backup:", err.message);
    questions = getBackupQuestions();
  }

  // Hide loader, show first question
  document.getElementById("aiLoader").style.display    = "none";
  document.getElementById("quizContent").style.display = "block";
  displayQuestion(0);
}
```

**How the AI Agent works step by step:**
1. User submits form → `generateQuestionsFromAI()` is called
2. JavaScript sends an HTTP POST request to Google's Gemini server
3. The request includes a prompt: "Generate 10 AWS questions as JSON"
4. Gemini AI processes the prompt (2-3 seconds)
5. Gemini returns a JSON array with 10 questions, 4 options each, and correct answer index
6. JavaScript parses the JSON and stores questions in the `questions` array
7. Questions are displayed one by one on screen

### 5.3 server.js — Node.js Backend (Optional)

Used when running on localhost. Keeps the Gemini API key secure on the server side.

```javascript
const express = require('express');
const https   = require('https');   // built into Node.js
const path    = require('path');

const app  = express();
const PORT = 3000;

// Your Gemini API key stored SAFELY on server (not visible to users)
const GEMINI_API_KEY = "YOUR_GEMINI_API_KEY_HERE";

// Serve all frontend files (index.html, style.css, app.js)
app.use(express.static(path.join(__dirname)));

// API endpoint that app.js calls when running on localhost
app.post('/api/generate-questions', (req, res) => {
  // Build the request to Gemini API
  const requestBody = JSON.stringify({
    contents: [{ parts: [{ text: "Generate 10 AWS exam questions..." }] }]
  });

  // Make HTTPS request to Google's Gemini API
  const options = {
    hostname: 'generativelanguage.googleapis.com',
    path: `/v1beta/models/gemini-2.0-flash:generateContent?key=${GEMINI_API_KEY}`,
    method: 'POST',
    headers: { 'Content-Type': 'application/json' }
  };

  const apiReq = https.request(options, (apiRes) => {
    let data = '';
    apiRes.on('data', chunk => data += chunk);
    apiRes.on('end', () => {
      const parsed    = JSON.parse(data);
      const aiText    = parsed.candidates[0].content.parts[0].text;
      const questions = JSON.parse(aiText.replace(/```json|```/g,'').trim());
      res.json({ success: true, questions });
    });
  });

  apiReq.write(requestBody);
  apiReq.end();
});

app.listen(PORT, () => {
  console.log(`✅ Server running at http://localhost:${PORT}`);
});
```

**Why server.js?**
When you open index.html directly (file://) the Gemini API key is visible in the browser's source code. Using server.js keeps the key hidden on your computer — only the server knows it, not the browser.

### Error 4: Certificate Text Overlapping — Signatures on top of description

**What happened:** The certificate PDF showed Morgan Willis and Rudy Chetty signatures overlapping the description text and course name.

**Root Cause:** The Y positions of certificate elements were calculated dynamically based on name length. When the name was short, gaps above were calculated correctly, but when using dynamic positioning for subsequent elements, the signatures ended up too high.

**Solution:** Changed signatures to use **fixed absolute Y position** from the top of the canvas, independent of other elements.

```javascript
// BEFORE — dynamic Y (caused overlapping)
const sigY = afterName + px(100);  // relative to name position — WRONG

// AFTER — fixed Y position (never moves)
const SIG_Y = px(183);   // always 183mm from top — CORRECT
```

This means regardless of how long or short the user's name is, the signatures always appear at the same fixed location near the bottom.

### Error 5: Seal Arc Text Reversed / Scrambled

**What happened:** The circular stamp on the certificate showed "BUILDING A" and "BETTER FUTURE" text appearing scrambled, reversed, or rotated incorrectly.

**Root Cause:** The arc text rotation formula was wrong. When placing characters along a circle, the rotation angle for each character must be:
- **Top arc:** `rotate(angle - 90°)` → character tops point AWAY from center (outward/upward)
- **Bottom arc:** `rotate(angle + 90°)` → character tops point DOWNWARD/outward

The original code used the same formula for both top and bottom arcs, causing bottom arc text to appear reversed.

**Solution:**
```javascript
function arcText(text, cx, cy, r, fromDeg, toDeg, outsideUp) {
  const chars = text.split("");
  chars.forEach((ch, i) => {
    const ang = startRad + (i/(n-1)) * (endRad-startRad);
    const lx  = cx + r * Math.cos(ang);
    const ly  = cy + r * Math.sin(ang);
    ctx.save();
    ctx.translate(lx, ly);

    if (outsideUp) {
      ctx.rotate(ang - Math.PI/2);  // TOP ARC — letters point upward
    } else {
      ctx.rotate(ang + Math.PI/2);  // BOTTOM ARC — letters point downward
    }

    ctx.fillText(ch, 0, 0);
    ctx.restore();
  });
}

// Top arc: outsideUp = true  (letters readable from outside, top of circle)
arcText("BUILDING  A",     SX, SY, arcR, 225, 315, true);

// Bottom arc: outsideUp = false (letters readable from outside, bottom of circle)
arcText("BETTER  FUTURE",  SX, SY, arcR, 45,  135, false);
```

### Error 6: Certificate Font Sizes Too Big — Text Going Outside Card Boundaries

**What happened:** The "CERTIFICATE OF COMPLETION" heading and "AWS CLOUD PRACTITIONER ESSENTIAL" text were overflowing outside the white card area into the blue border strips.

**Root Cause:** Fixed pixel font sizes (like `font-size: 58px`) that were correct for one canvas size became too large when the canvas dimensions changed.

**Solution:** Added **auto-shrink loops** that reduce font size until the text fits within the card width.

```javascript
// BEFORE — fixed font (overflows on some canvases)
ctx.font = "bold 58px Arial";
ctx.fillText("CERTIFICATE OF COMPLETION", MID, 242);

// AFTER — auto-shrink until it fits
let titleFs = 56;
ctx.font = `bold ${titleFs}px Arial, sans-serif`;
while (ctx.measureText("CERTIFICATE OF COMPLETION").width > CW - 120) {
  titleFs--;  // reduce by 1px until it fits
  ctx.font = `bold ${titleFs}px Arial, sans-serif`;
}
ctx.fillText("CERTIFICATE OF COMPLETION", MID, 242);
```

The `ctx.measureText(text).width` method returns the actual pixel width of the text at the current font size, allowing us to compare it to the available card width and shrink accordingly.

### Error 7: Video Audio Not Playing

**What happened:** When the user clicked Play on a video, the progress bar moved but no audio was heard.

**Root Cause:** The Web Speech API requires voices to be loaded first. In some browsers, voices are not immediately available — `speechSynthesis.getVoices()` returns an empty array until the `voiceschanged` event fires.

**Solution:** Added a voice pre-loading listener at page initialization:

```javascript
// In window.onload — pre-load voices before user starts video
window.onload = function() {
  updateProgress();

  // Pre-load available voices
  if (speechSynthesis.onvoiceschanged !== undefined) {
    speechSynthesis.onvoiceschanged = () => speechSynthesis.getVoices();
  }
};

// Then in speakNextLine() — try to pick best English voice
function speakNextLine() {
  const utterance = new SpeechSynthesisUtterance(line);
  const voices    = speechSynthesis.getVoices();

  // Prefer Google or Microsoft English voices for clarity
  const preferred = voices.find(v =>
    (v.name.includes("Google") || v.name.includes("Microsoft")) &&
    v.lang.startsWith("en")
  );
  if (preferred) utterance.voice = preferred;

  speechSynthesis.speak(utterance);
}
```

### Error 8: Certificate Showing "AWS Candidate" Instead of User's Name

**What happened:** The certificate downloaded with "AWS Candidate" printed instead of the actual user's name entered in the form.

**Root Cause:** The `userName` global variable was not being saved when the form was submitted. The certificate function was using a hardcoded string instead of the variable.

**Solution:**
```javascript
// In submitForm() — save name to global variable
function submitForm() {
  const name = document.getElementById("inp-name").value.trim();
  userName = name;  // ← Save to global variable
  // ...
}

// In downloadCertificate() — use the global variable
const certHTML = `...
  <div class="cert-name">${userName}</div>  // ← Uses saved name
...`;
```
